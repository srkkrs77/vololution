<?php
$tvName = $params['tv'] ?? '';
if ($tvName === '') return '';

$raw = $modx->getTemplateVar($tvName, '*', $modx->documentIdentifier);
if (empty($raw['value'])) return '';

$data = json_decode($raw['value'], true);
if (!is_array($data) || empty($data[0])) return '';

$row = $data[0];
$src = htmlspecialchars($row['image'] ?? '', ENT_QUOTES, 'UTF-8');
$alt = htmlspecialchars($row['alt']   ?? '', ENT_QUOTES, 'UTF-8');

if ($src === '') return '';

return '<img src="' . $src . '" alt="' . $alt . '">';
